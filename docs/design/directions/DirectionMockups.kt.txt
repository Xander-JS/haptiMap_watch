private data class Pal(
    val bg: Color, val surface: Color, val onBg: Color, val muted: Color,
    val primary: Color, val onPrimary: Color, val tour: Color, val onTour: Color,
)

private enum class Nav { PULSE, SIGN, HALO }

private data class Dir(
    val light: Pal,
    val dark: Pal,
    val display: FontFamily,
    val displayWeight: FontWeight,
    val distanceSize: TextUnit,
    val corner: Dp,
    val outlineIcons: Boolean,
    val nav: Nav,
    val capsLabels: Boolean,
)

private val Condensed = FontFamily(Typeface(android.graphics.Typeface.create("sans-serif-condensed", android.graphics.Typeface.BOLD)))

    outlineIcons = true, nav = Nav.HALO, capsLabels = false,

private object FixedTime : TimeSource {
    @Composable override fun currentTime() = "10:10"
}

@Composable
private fun Watch(d: Dir, content: @Composable BoxScope.(Pal) -> Unit) {
    val p = if (isSystemInDarkTheme()) d.dark else d.light
    MaterialTheme(colorScheme = ColorScheme(primary = p.primary, onPrimary = p.onPrimary, background = p.bg, onBackground = p.onBg, onSurface = p.onBg)) {
        Box(Modifier.fillMaxSize().background(p.bg)) {
            content(p)
            TimeText(timeSource = FixedTime) { timeTextCurvedText(it, CurvedTextStyle(color = p.onBg)) }
        }
    }
}

private fun Dir.label(text: String) = if (capsLabels) text.uppercase() else text

@Composable
private fun EdgeAction(text: String, container: Color, content: Color, d: Dir) = Box(Modifier.fillMaxSize(), contentAlignment = Alignment.BottomCenter) {
    EdgeButton(onClick = {}, buttonSize = EdgeButtonSize.Medium, colors = ButtonDefaults.buttonColors(containerColor = container, contentColor = content)) {
        Text(d.label(text), fontWeight = FontWeight.Bold, fontSize = 15.sp, letterSpacing = if (d.capsLabels) 1.sp else 0.sp)
    }
}

// ---------- Navegación: próxima maniobra + distancia ----------

@Composable
private fun NavScreen(d: Dir, tour: Boolean) = Watch(d) { p ->
    val accent = if (tour) p.tour else p.primary
    val onAccent = if (tour) p.onTour else p.onPrimary
    val arrow = painterResource(if (d.outlineIcons) R.drawable.ic_turn_left_outline else R.drawable.ic_turn_left)
    when (d.nav) {
        Nav.PULSE -> Column(Modifier.align(Alignment.Center).padding(top = 18.dp), horizontalAlignment = Alignment.CenterHorizontally) {
            Box(Modifier.size(92.dp), contentAlignment = Alignment.Center) {
                Canvas(Modifier.fillMaxSize()) {
                    drawCircle(accent.copy(alpha = 0.18f), radius = size.minDimension / 2, style = Stroke(2.dp.toPx()))
                    drawCircle(accent.copy(alpha = 0.40f), radius = size.minDimension / 2 - 9.dp.toPx(), style = Stroke(2.dp.toPx()))
                }
                Box(Modifier.size(58.dp).background(accent, CircleShape), contentAlignment = Alignment.Center) {
                    Icon(arrow, null, tint = onAccent, modifier = Modifier.size(36.dp))
                }
            }
            Row(verticalAlignment = Alignment.Bottom) {
                Text("45", style = TextStyle(fontFamily = d.display, fontWeight = d.displayWeight, fontSize = d.distanceSize, lineHeight = d.distanceSize, color = p.onBg, fontFeatureSettings = "tnum"))
                Text(" m", style = TextStyle(fontFamily = d.display, fontWeight = FontWeight.Bold, fontSize = 20.sp, color = accent), modifier = Modifier.padding(bottom = 9.dp))
            }
            Text("MAIN ST", fontFamily = d.display, fontSize = 15.sp, letterSpacing = 1.5.sp, color = p.muted)
        }
        Nav.SIGN -> Column(Modifier.align(Alignment.Center).padding(top = 14.dp), horizontalAlignment = Alignment.CenterHorizontally) {
            Box(Modifier.size(66.dp).background(accent, RoundedCornerShape(d.corner)), contentAlignment = Alignment.Center) {
                Icon(arrow, null, tint = onAccent, modifier = Modifier.size(44.dp))
            }
            Spacer(Modifier.height(4.dp))
            Text("45 m", style = TextStyle(fontFamily = d.display, fontWeight = d.displayWeight, fontSize = d.distanceSize, letterSpacing = (-1.5).sp, color = p.onBg, fontFeatureSettings = "tnum"))
            Text("Main Street", fontWeight = FontWeight.SemiBold, fontSize = 16.sp, color = p.onBg)
            Spacer(Modifier.height(10.dp))
            // franja de sendero: el modo se lee como la marca de color de una ruta señalizada
            Box(Modifier.width(44.dp).height(5.dp).background(accent, RoundedCornerShape(3.dp)))
        }
        Nav.HALO -> {
            Canvas(Modifier.fillMaxSize().padding(5.dp)) {
                val w = 7.dp.toPx()
                val arcSize = Size(size.width - w, size.height - w)
                val topLeft = Offset(w / 2, w / 2)
                drawArc(p.surface, startAngle = -30f, sweepAngle = 240f, useCenter = false, topLeft = topLeft, size = arcSize, style = Stroke(w, cap = StrokeCap.Round))
                drawArc(accent, startAngle = 210f, sweepAngle = -240f * 0.72f, useCenter = false, topLeft = topLeft, size = arcSize, style = Stroke(w, cap = StrokeCap.Round))
            }
            Column(Modifier.align(Alignment.Center).padding(top = 10.dp), horizontalAlignment = Alignment.CenterHorizontally) {
                Icon(arrow, null, tint = accent, modifier = Modifier.size(56.dp))
                Text("45", style = TextStyle(fontFamily = d.display, fontWeight = d.displayWeight, fontSize = d.distanceSize, lineHeight = d.distanceSize, color = p.onBg, fontFeatureSettings = "tnum"))
                Text("metres · Main St", fontSize = 14.sp, color = p.muted)
            }
        }
    }
}

// ---------- Tarjeta de sugerencia POI (Tour) ----------

@Composable
private fun PoiScreen(d: Dir) = Watch(d) { p ->
    Column(Modifier.align(Alignment.Center).padding(bottom = 40.dp, start = 24.dp, end = 24.dp), horizontalAlignment = Alignment.CenterHorizontally) {
        Box(Modifier.size(68.dp), contentAlignment = Alignment.Center) {
            Canvas(Modifier.fillMaxSize()) {
                val w = 4.dp.toPx()
                drawArc(p.surface, 0f, 360f, false, Offset(w / 2, w / 2), Size(size.width - w, size.height - w), style = Stroke(w))
                drawArc(p.tour, -90f, 360f * 0.6f, false, Offset(w / 2, w / 2), Size(size.width - w, size.height - w), style = Stroke(w, cap = StrokeCap.Round))
            }
            val shape = if (d.nav == Nav.SIGN) RoundedCornerShape(d.corner) else CircleShape
            Box(Modifier.size(48.dp).background(if (d.outlineIcons) Color.Transparent else p.tour, shape), contentAlignment = Alignment.Center) {
                Icon(painterResource(R.drawable.ic_exploration_museum), null, tint = if (d.outlineIcons) p.tour else p.onTour, modifier = Modifier.size(28.dp))
            }
        }
        Spacer(Modifier.height(6.dp))
        Text("City Museum", fontFamily = if (d.capsLabels) d.display else FontFamily.SansSerif, fontWeight = FontWeight.Bold, fontSize = 20.sp, color = p.onBg, textAlign = TextAlign.Center)
        Text("+3 min", fontWeight = FontWeight.SemiBold, fontSize = 16.sp, color = p.tour)
    }
    EdgeAction("Explore", p.tour, p.onTour, d)
}

// ---------- Selector de perfil (Focus / Tour) ----------

@Composable
private fun ProfileScreen(d: Dir) = Watch(d) { p ->
    Column(Modifier.align(Alignment.Center).padding(horizontal = 22.dp).padding(top = 4.dp), horizontalAlignment = Alignment.CenterHorizontally, verticalArrangement = Arrangement.spacedBy(8.dp)) {
        Text(d.label("Choose mode"), fontFamily = if (d.capsLabels) d.display else FontFamily.SansSerif, fontWeight = FontWeight.Bold, fontSize = 15.sp, letterSpacing = if (d.capsLabels) 1.5.sp else 0.sp, color = p.muted)
        ModeOption(d, p, R.drawable.ic_profile_focus, "Focus", "Straight there", p.primary, p.onPrimary, selected = true)
        ModeOption(d, p, R.drawable.ic_profile_tour, "Tour", "Discover places", p.tour, p.onTour, selected = false)
    }
}

@Composable
private fun ModeOption(d: Dir, p: Pal, icon: Int, title: String, sub: String, accent: Color, onAccent: Color, selected: Boolean) {
    val shape = RoundedCornerShape(d.corner)
    val base = Modifier.fillMaxWidth().height(54.dp).clip(shape)
    val mod = if (selected) base.background(accent) else base.background(p.surface).let { if (d.nav == Nav.SIGN) it.border(2.dp, accent, shape) else it }
    val fg = if (selected) onAccent else p.onBg
    Row(mod.padding(horizontal = 12.dp), verticalAlignment = Alignment.CenterVertically) {
        Box(Modifier.size(36.dp).background(if (selected) onAccent.copy(alpha = 0.14f) else accent.copy(alpha = 0.18f), CircleShape), contentAlignment = Alignment.Center) {
            Icon(painterResource(icon), null, tint = if (selected) onAccent else accent, modifier = Modifier.size(22.dp))
        }
        Spacer(Modifier.width(10.dp))
        Column {
            Text(d.label(title), fontFamily = if (d.capsLabels) d.display else FontFamily.SansSerif, fontWeight = FontWeight.Bold, fontSize = 17.sp, letterSpacing = if (d.capsLabels) 1.sp else 0.sp, color = fg)
            Text(sub, fontSize = 12.sp, color = if (selected) onAccent.copy(alpha = 0.8f) else p.muted, maxLines = 1)
        }
    }
}

// ---------- Búsqueda ----------

@Composable
private fun SearchScreen(d: Dir) = Watch(d) { p ->
    Column(Modifier.align(Alignment.Center).padding(top = 16.dp), horizontalAlignment = Alignment.CenterHorizontally, verticalArrangement = Arrangement.spacedBy(10.dp)) {
        Text(d.label("Where to?"), fontFamily = if (d.capsLabels) d.display else FontFamily.SansSerif, fontWeight = FontWeight.Bold, fontSize = if (d.capsLabels) 17.sp else 19.sp, letterSpacing = if (d.capsLabels) 1.5.sp else 0.sp, color = p.onBg)
        // la voz es la acción principal: un solo objetivo grande, sin texto
        val voiceShape = if (d.nav == Nav.SIGN) RoundedCornerShape(d.corner) else CircleShape
        Box(Modifier.size(64.dp).background(p.primary, voiceShape), contentAlignment = Alignment.Center) {
            Icon(painterResource(R.drawable.ic_mic), "Search by voice", tint = p.onPrimary, modifier = Modifier.size(32.dp))
        }
        // categorías como iconos: se reconocen sin leer; el nombre queda para TalkBack
        Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
            listOf(R.drawable.ic_category_cafe, R.drawable.ic_category_pharmacy, R.drawable.ic_category_store).forEach {
                val s = if (d.nav == Nav.SIGN) RoundedCornerShape(10.dp) else CircleShape
                Box(Modifier.size(44.dp).background(p.surface, s), contentAlignment = Alignment.Center) {
                    Icon(painterResource(it), null, tint = p.onBg, modifier = Modifier.size(22.dp))
                }
            }
        }
    }
}
